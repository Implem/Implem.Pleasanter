public static string OpenBulkUpdateSelectorDialog(Context context, SiteSettings ss)
{
    if (!context.CanUpdate(ss: ss))
    {
        return Messages.ResponseHasNotPermission(context: context).ToJson();
    }
    var where = SelectedWhere(
        context: context,
        ss: ss);
    if (where == null)
    {
        return Messages.ResponseSelectTargets(context: context).ToJson();
    }
    var optionCollection = ss.GetAllowBulkUpdateOptions(context: context);
    var target = optionCollection.FirstOrDefault().Key;
    var hb = new HtmlBuilder();
    return new ResponseCollection(context: context)
        .Html(
            "#BulkUpdateSelectorDialog",
            hb.BulkUpdateSelectorDialog(
                context: context,
                ss: ss,
                optionCollection: optionCollection,
                action: () => hb
                    .BulkUpdateEditor(
                        context: context,
                        ss: ss,
                        target: target)))
        .ToJson();
}

public static string BulkUpdate(Context context, SiteSettings ss)
{
    var resultData = BulkUpdateCore(
        context: context,
        ss: ss);
    switch (resultData.ErrorData?.Type)
    {
        case null:
        case Error.Types.None:
            break;
        case Error.Types.HasNotPermission:
            return Messages.ResponseHasNotPermission(context: context).ToJson();
        case Error.Types.SelectTargets:
            return Messages.ResponseSelectTargets(context: context).ToJson();
        case Error.Types.ImportLock:
            return Error.Types.ImportLock.MessageJson(context: context);
        default:
            return resultData.ErrorData.MessageJson(context: context);
    }
    return GridRows(
        context: context,
        ss: ss,
        clearCheck: true,
        message: Messages.BulkUpdated(
            context: context,
            data: new string[]
            {
                ss.Title,
                (resultData.SuccessCount + resultData.FailureCount).ToString()
            }));
}

public static BulkUpdateResultData BulkUpdateCore(
    Context context,
    SiteSettings ss)
{
    var validated = ValidateBulkUpdate(
        context: context,
        ss: ss);
    if (validated.ErrorData.Type != Error.Types.None)
    {
        return new BulkUpdateResultData
        {
            ErrorData = validated.ErrorData
        };
    }
    var columns = validated.Columns;
    var where = validated.Where;
    var #modelName#Model = new #ModelName#Model(
        context: context,
        ss: ss,
        #modelName#Id: 0,
        formData: context.Forms);
    using var exclusiveObj = new Sessions.TableExclusive(context: context);
    if (!exclusiveObj.TryLock())
    {
        return new BulkUpdateResultData
        {
            ErrorData = new ErrorData(type: Error.Types.ImportLock)
        };
    }
    var resultData = BulkUpdate(
        context: context,
        ss: ss,
        columns: columns,
        where: where,
        watchdog: () => exclusiveObj.Refresh());
    var data = new string[]
    {
        ss.Title,
        (resultData.SuccessCount + resultData.FailureCount).ToString()
    };
    ss.Notifications
        .Where(o => o.MonitorChangesColumns.Any(columnName => columns.Any(q => q.ColumnName == columnName)))
        .ForEach(notification =>
        {
            var body = new System.Text.StringBuilder();
            body.Append(Locations.ItemIndexAbsoluteUri(
                context: context,
                ss.SiteId) + "\n");
            body.Append(
                $"{Displays.#TableName#_Updator(context: context)}: ",
                $"{context.User.Name}\n");
            columns.ForEach(column =>
                body.Append(
                    $"{column.LabelText} : ",
                    $"{#modelName#Model.ToDisplay(context: context, ss: ss, column: column, mine: null)}\n"));
            if (notification.AfterBulkUpdate != false)
            {
                notification.Send(
                    context: context,
                    ss: ss,
                    title: Displays.BulkUpdated(
                        context: context,
                        data: data),
                    body: body.ToString());
            }
        });
    return resultData;
}

private static BulkUpdateResultData BulkUpdate(
    Context context,
    SiteSettings ss,
    List<Column> columns,
    SqlWhereCollection where,
    Action watchdog = null)
{
    var onBulkUpdatingExtendedStatements = new List<SqlStatement>().OnBulkUpdatingExtendedSqls(
        context: context,
        siteId: ss.SiteId);
    if (onBulkUpdatingExtendedStatements.Count > 0)
    {
        watchdog?.Invoke();
        Repository.ExecuteNonQuery(
            context: context,
            transactional: true,
            statements: onBulkUpdatingExtendedStatements.ToArray());
    }
    var successCount = 0;
    var failureCount = 0;
    var bulkUpdateFormKeys = columns
        .Select(column => $"{ss.ReferenceType}_{column.ColumnName}")
        .ToHashSet();
    var formData = context.Forms
        .Where(o => bulkUpdateFormKeys.Contains(o.Key))
        .ToDictionary(o => o.Key, o => o.Value);
    new #ModelName#Collection(
        context: context,
        ss: ss,
        join: ss.Join(
            context: context,
            join: new IJoin[]
            {
                where
            }),
        where: where).ForEach(#modelName#Model =>
        {
            watchdog?.Invoke();
            #modelName#Model.SetByForm(
                context: context,
                ss: ss,
                formData: formData);
            #modelName#Model.SetBySettings(
                context: context,
                ss: ss);
            #modelName#Model.VerUp = Versions.MustVerUp(
                context: context,
                ss: ss,
                baseModel: #modelName#Model);
            var errorData = #modelName#Model.Update(
                context: context,
                ss: ss,
<!--Model_Utilities_AiConnectSyncSuppression ( Indent = 6 ) -->
                extendedSqls: false);
            if (errorData.Type == Error.Types.None)
            {
                successCount++;
            }
            else
            {
                failureCount++;
            }
        });
    var onBulkUpdatedExtendedStatements = new List<SqlStatement>().OnBulkUpdatedExtendedSqls(
        context: context,
        siteId: ss.SiteId);
    if (onBulkUpdatedExtendedStatements.Count > 0)
    {
        watchdog?.Invoke();
        Repository.ExecuteNonQuery(
            context: context,
            transactional: true,
            statements: onBulkUpdatedExtendedStatements.ToArray());
    }
    return new BulkUpdateResultData
    {
        SuccessCount = successCount,
        FailureCount = failureCount
    };
}

public static int BulkUpdateTargetCount(
    Context context,
    SiteSettings ss)
{
    var selectedWhere = SelectedWhere(
        context: context,
        ss: ss);
    if (selectedWhere == null)
    {
        return 0;
    }
    var recordSelector = new RecordSelector(context: context);
    if (!recordSelector.All)
    {
        return recordSelector.Selected.Count;
    }
    var view = Views.GetBySession(
        context: context,
        ss: ss);
    var where = view.Where(
        context: context,
        ss: ss,
        where: selectedWhere,
        itemJoin: false);
    var param = view.Param(
        context: context,
        ss: ss);
    return Repository.ExecuteScalar_int(
        context: context,
        statements: Rds.Select#TableName#(
            column: Rds.#TableName#Column().#TableName#Count(),
            join: ss.Join(
                context: context,
                join: new IJoin[]
                {
                    where
                }),
            where: where,
            param: param));
}

public static ErrorData ValidateBeforeEnqueueBulkUpdate(
    Context context,
    SiteSettings ss)
{
    return ValidateBulkUpdate(
        context: context,
        ss: ss).ErrorData;
}

private static (ErrorData ErrorData, List<Column> Columns, SqlWhereCollection Where) ValidateBulkUpdate(
    Context context,
    SiteSettings ss)
{
    if (!context.CanUpdate(ss: ss))
    {
        return (new ErrorData(type: Error.Types.HasNotPermission), null, null);
    }
    var target = context.Forms.Data("BulkUpdateColumnName");
    var columns = ss.GetBulkUpdateColumns(
        context: context,
        target: target);
    ss.SetBulkUpdateColumnDetail(
        columns: columns,
        target: target);
    columns = columns
        .Where(column => column.EditorReadOnly != true)
        .ToList();
    if (!columns.Any())
    {
        return (new ErrorData(type: Error.Types.InvalidRequest), null, null);
    }
    if (!columns.All(column => column.CanUpdate(
        context: context,
        ss: ss,
        mine: null)))
    {
        return (new ErrorData(type: Error.Types.HasNotPermission), null, null);
    }
    var selectedWhere = SelectedWhere(
        context: context,
        ss: ss);
    if (selectedWhere == null)
    {
        return (new ErrorData(type: Error.Types.SelectTargets), null, null);
    }
    var view = Views.GetBySession(
        context: context,
        ss: ss);
    var where = view.Where(
        context: context,
        ss: ss,
        where: selectedWhere,
        itemJoin: false);
    var param = view.Param(
        context: context,
        ss: ss,
        param: new Rds.#TableName#ParamCollection());
    var invalid = ExistsLockedRecord(
        context: context,
        ss: ss,
        where: where,
        param: param,
        orderBy: view.OrderBy(
            context: context,
            ss: ss),
        lockedColumn: columns.Any(o => o.ColumnName == "Locked"));
    if (invalid.Type != Error.Types.None)
    {
        return (invalid, null, null);
    }
    return (invalid, columns, where);
}
